//Ожидаемый результат:

namespace Lesson14
{
    public abstract class Transport
    {
        public string Model { get; set; }
        protected int Speed = 0;

        public void ShowInfo()
        {
            Console.WriteLine($"Модель: {this.Model}, скорость: {this.Speed} км/ч");
        }

        public virtual void Move()
        {
            Console.WriteLine("Траспорт движется");
        }

    }

    public class Car : Transport
    {
        public void Accelerate(int value)
        {
            if (base.Speed + value >= 200)
            {
                Console.WriteLine("Слишком большая скорость");
            }
            else
            {
                base.Speed += value;
            }
        }

        public override void Move()
        {
            Console.WriteLine("Машина едет по дороге");
        }
    }

    public class Bicycle : Transport
    {
        public void Pedal()
        {
            base.Speed += 5;
        }

        public override void Move()
        {
            Console.WriteLine("Велосипед крутит педали");
        }
    }
    class Program
    {
        public static void Main(string[] args)
        {
            var car = new Car { Model = "Audi" }; 
            car.Accelerate(100); 
            car.ShowInfo();
            car.Move(); 
            var bike = new Bicycle { Model = "Cube" }; 
            bike.Pedal();
            bike.ShowInfo();
            bike.Move();
        }
    }
}
